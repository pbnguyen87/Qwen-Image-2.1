# Dependencies for generate.py (text-to-image and image editing).
#
#   pip install -r requirements.txt
#
# torch must match your CUDA. The plain wheel on PyPI is built for one CUDA
# version only, so install it first from the index that matches your driver,
# then run this file:
#
#   pip install torch torchvision --index-url https://download.pytorch.org/whl/cu128
#   pip install -r requirements.txt
#
# nvidia-smi shows the CUDA version the driver supports; cu128 suits CUDA 12.8,
# cu126 suits 12.6. On Apple Silicon or CPU, plain `pip install torch` is right.

# Lower bounds, per the model card. torch>=2.4 is the documented floor; the
# QwenImage21Pipeline needs transformers 5.17 or newer for the Qwen3-VL text
# encoder the checkpoint ships with.

transformers>=5.17
accelerate>=1.0
